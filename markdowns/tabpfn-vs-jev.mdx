# TabPFN vs Jev: phishing, Kickstarter and job fraud

This guide aims to find out - when it is better to use Jev and when Jev needs a tool like TabPFN to make grounded decisions. We do that by comparing Jev and TabPFN on three datasets of different kind - one with mostly numeric data, one with mostly text, and one fairly mixed between text, categories and numbers. We use the same training and test rows with same random seeds, maxing out Jev's context, and also compare TabPFN performance with more training data on Kickstarter and job fraud, as TabPFN's context can fit up to 1M rows.

1. **Phishing websites:** Mostly numeric / yes & no data describing the websites. 350 training rows, 300 test rows.
2. **Kickstarter funding:** Well-known dataset with mostly text-based features. 150 training rows, 300 test rows. We also test TabPFN with 1,000 and 5,000 training rows.
3. **Job posting fraud:** Fairly mixed dataset with both text and categories contributing to the result. Both models are tested with 44 training rows and 300 test rows, followed by TabPFN with 300, 1,000 and 5,000 training rows.

# A TLDR of what we found

- **Websites:** TabPFN gets 93.7% right versus Jev's 73.0%, with smaller probability errors.
- **Kickstarter:** with 150 examples each, Jev ranks successes better. TabPFN benefits from more labels: at 5,000 rows its AUROC reaches 0.735, versus Jev's 0.725.
- **Job fraud:** TabPFN with 5,000 examples and validation-selected tuning catches 13 of 15 fraud cases with one false alarm. Jev with 44 examples catches 8 with 24 false alarms after tuning. The extra-data result supports testing TabPFN as Jev's prediction tool; it is not an equal-data model comparison.

### Reading the results

Accuracy is the fraction of correct decisions. Recall is the fraction of positive cases caught; F1 balances recall with precision (how many flagged cases are actually positive). These use a 0.5 threshold, except in the final section, which selects thresholds using training data.

AUROC and average precision measure how well the model ranks positive cases above negative ones; higher is better. Average precision is especially useful when fraud is rare. **Brier score** is the average squared error of the probabilities; **log loss** penalizes confident mistakes more heavily. Both are better when lower, and neither is an error percentage.

In the calibration plots, points near the diagonal mean predicted risks match observed rates. Five bins, counts and 95% error bars show how much evidence supports each point; 300 test rows give only a rough picture.

```python
def evaluate(y_true, predictions):
    rows = []

    for name, probabilities in predictions.items():
        labels = probabilities >= 0.5
        rows.append({
            "Model": name,
            "Accuracy": accuracy_score(y_true, labels),
            "F1": f1_score(y_true, labels, zero_division=0),
            "Recall": recall_score(y_true, labels, zero_division=0),
            "AUROC": roc_auc_score(y_true, probabilities),
            "Average precision": average_precision_score(y_true, probabilities),
            "Log loss": log_loss(y_true, probabilities, labels=[0, 1]),
            "Brier": brier_score_loss(y_true, probabilities),
        })

    return pd.DataFrame(rows).set_index("Model")


def plot_results(y_true, predictions, title):
    names = list(predictions)
    scores = evaluate(y_true, predictions)
    y_true = np.asarray(y_true)

    # Put the measured differences first, with the same scale for both models.
    fig, axes = plt.subplots(1, 4, figsize=(14, 3.6), layout="constrained")
    for ax, metric in zip(axes, ["Accuracy", "AUROC", "Brier", "Log loss"]):
        values = scores.loc[names, metric]
        bars = ax.bar(names, values, color=[colors[name] for name in names], width=0.6)
        labels = [f"{value:.0%}" if metric == "Accuracy" else f"{value:.3f}" for value in values]
        ax.bar_label(bars, labels=labels, padding=6, fontsize=12, fontweight="bold")
        ax.set_ylim(0, 1.15 if metric in ["Accuracy", "AUROC"] else values.max() * 1.35)
        ax.set_title(metric + (" ↑" if metric in ["Accuracy", "AUROC"] else " ↓"), loc="left")
        ax.set_axisbelow(True)
        ax.yaxis.grid(color="#EEF0F4")
        ax.tick_params(length=0)
        ax.spines[["left", "bottom"]].set_visible(False)

    fig.suptitle(title, color="#101075", fontsize=16, fontweight="bold")
    plt.show()

    # Label ranking curves with their scores rather than relying on visual distance.
    fig, axes = plt.subplots(1, 2, figsize=(11, 4.4), layout="constrained")
    for name, probabilities in predictions.items():
        fpr, tpr, _ = roc_curve(y_true, probabilities)
        precision, recall, _ = precision_recall_curve(y_true, probabilities)
        axes[0].plot(
            fpr, tpr, color=colors[name], linewidth=2.5,
            label=f"{name} · AUROC {scores.loc[name, 'AUROC']:.3f}",
        )
        axes[1].step(
            recall, precision, where="post", color=colors[name], linewidth=2.5,
            label=f"{name} · AP {scores.loc[name, 'Average precision']:.3f}",
        )

    axes[0].plot([0, 1], [0, 1], "--", color="#C4CAD4", label="Random ranking")
    axes[1].axhline(
        y_true.mean(), linestyle="--", color="#C4CAD4",
        label=f"Positive-class prevalence: {y_true.mean():.0%}",
    )
    axes[0].set(title="ROC", xlabel="False positive rate", ylabel="True positive rate")
    axes[1].set(title="Precision–recall", xlabel="Recall", ylabel="Precision")
    for ax in axes:
        ax.set_xlim(-0.02, 1.02)
        ax.set_ylim(-0.02, 1.02)
        ax.grid(color="#EEF0F4")
        ax.legend(fontsize=9, loc="lower right" if ax is axes[0] else "lower left")
    plt.show()

    # Separate calibration panels avoid overlapping lines and expose sparse bins.
    fig, axes = plt.subplots(
        2, 2, figsize=(11, 7.4), layout="constrained",
        gridspec_kw={"height_ratios": [2.4, 1]},
    )
    largest_bin = max(np.histogram(p, bins=np.linspace(0, 1, 11))[0].max() for p in predictions.values())
    for column, (name, probabilities) in enumerate(predictions.items()):
        probabilities = np.asarray(probabilities)
        bins = pd.DataFrame({"probability": probabilities, "outcome": y_true})
        bins["bin"] = pd.cut(bins["probability"], bins=np.linspace(0, 1, 6), include_lowest=True)
        points = bins.groupby("bin", observed=True).agg(
            predicted=("probability", "mean"),
            observed=("outcome", "mean"),
            count=("outcome", "size"),
        )

        # Wilson intervals describe uncertainty in each observed event rate.
        n = points["count"].to_numpy()
        rate = points["observed"].to_numpy()
        z = 1.96
        denominator = 1 + z**2 / n
        center = (rate + z**2 / (2 * n)) / denominator
        half_width = z * np.sqrt(rate * (1 - rate) / n + z**2 / (4 * n**2)) / denominator
        errors = np.maximum(0, [rate - (center - half_width), center + half_width - rate])

        ax = axes[0, column]
        ax.plot([0, 1], [0, 1], "--", color="#B9C1CE", linewidth=1.5)
        ax.errorbar(
            points["predicted"], rate, yerr=errors, fmt="none",
            color=colors[name], alpha=0.65, capsize=4, linewidth=1.5,
        )
        ax.scatter(
            points["predicted"], rate, s=35 + 4 * n,
            color=colors[name], edgecolor="white", linewidth=1, zorder=3,
        )
        for predicted, observed, count in zip(points["predicted"], rate, n):
            ax.annotate(
                f"n={count}", (predicted, observed),
                xytext=(7 if predicted < 0.8 else -7, 10 if observed < 0.9 else -18),
                textcoords="offset points", ha="left" if predicted < 0.8 else "right",
                fontsize=9, color="#46505F",
            )
        ax.set(
            title=name, xlabel="Mean predicted probability", ylabel="Observed positive fraction",
            xlim=(-0.04, 1.04), ylim=(-0.05, 1.05),
        )
        ax.grid(color="#EEF0F4")

        ax = axes[1, column]
        ax.hist(
            [probabilities[y_true == 0], probabilities[y_true == 1]],
            bins=np.linspace(0, 1, 11), stacked=True,
            color=["#D5DCE6", "#101075"], edgecolor="white",
            label=["Actual negative", "Actual positive"],
        )
        ax.set(
            xlabel="Predicted probability", ylabel="Test rows",
            xlim=(0, 1), ylim=(0, largest_bin * 1.15),
        )
        ax.legend(fontsize=8, loc="upper center")
        ax.set_axisbelow(True)
        ax.yaxis.grid(color="#EEF0F4")

    fig.suptitle(
        "Calibration and probability distribution\n"
        "Same five probability bins · marker size = count · error bars = 95% Wilson intervals",
        color="#101075", fontsize=13, fontweight="bold",
    )
    plt.show()


def replace_test_duplicates(X, y, X_train, X_test, y_test):
    """Replace test rows seen in training; preserve test size and class counts."""
    training_features = pd.MultiIndex.from_frame(X_train)
    overlap = pd.MultiIndex.from_frame(X_test).isin(training_features)
    kept = X_test.loc[~overlap].copy()
    candidates = X.drop(index=X_train.index.union(X_test.index))
    excluded = training_features.append(pd.MultiIndex.from_frame(kept))
    candidates = candidates.loc[~pd.MultiIndex.from_frame(candidates).isin(excluded)]
    candidates = candidates.drop_duplicates()
    for label, count in y_test.value_counts().items():
        needed = int(count - (y.loc[kept.index] == label).sum())
        extra = candidates.loc[y.loc[candidates.index] == label].sample(
            n=needed, random_state=RANDOM_STATE,
        )
        kept = pd.concat([kept, extra])
    assert not pd.MultiIndex.from_frame(kept).isin(training_features).any()
    assert len(kept) == len(X_test)
    print(f"Replaced {int(overlap.sum())} test rows matching training features.")
    return kept, y.loc[kept.index]
```

## Part 1 — Phishing websites

### Download and split

[OpenML PhishingWebsites](https://www.openml.org/search?type=data&id=4534) contains 11,055 rows and 30 categorical indicators extracted from websites. The original target is −1 for phishing and +1 for legitimate; we encode phishing as the positive class.

We parse the dataset's categorical codes as integers and take a stratified random sample. The remaining rows are unused.

Replace any test row whose full feature vector occurs in training with a randomly sampled row of the same class. Both models use this same 300-row test set; feature values are unchanged.

```python
web_url = "https://openml.org/data/v1/download/1798106/PhishingWebsites.arff"
```

### TabPFN Plus

Pass the feature table and training labels directly to the client. All website columns are categorical codes. We use TabPFN 3.5 Plus with default settings.

In `tabpfn-client`, `create_default_for_version("v3.5")` selects the hosted **TabPFN-3.5-Plus**, as documented in [model selection](https://docs.priorlabs.ai/models/selecting-model-version). The local `tabpfn` package is a different interface.

```python
web_model = TabPFNClassifier.create_default_for_version(
    "v3.5",
    n_estimators=8,
    random_state=RANDOM_STATE,
    categorical_features_indices=list(range(X_web_train.shape[1])),
)

web_model.fit(X_web_train, y_web_train)
web_tabpfn = web_model.predict_proba(X_web_test)[:, list(web_model.classes_).index(1)]
```

## Part 2 — Kickstarter funding

We select eight feature columns and use their values exactly as loaded. No quote removal, date repair, outcome-tag removal, text normalization or imputation is applied. `disable_communication` is not a feature in this comparison because it may reflect the actual outcome.

The export contains malformed timestamps near 1970 and some text can reveal outcomes, which we allow for the purpose of this comparison.

#### Calibrated probabilities + F1 threshold

For both models, fit a temperature to minimize training-only validation log loss, then choose a threshold that maximizes validation F1. F1 balances precision and recall; it does not simply maximize fraud caught. Test labels are used only for evaluation.

```python
from sklearn.model_selection import StratifiedKFold

jobs_oof_predictions = {}

for training_size in [44, 300, 1_000, 5_000]:
    X_fit, y_fit = jobs_training_sets[training_size]
    n_folds = min(5, int(y_fit.value_counts().min()))
    folds = StratifiedKFold(n_splits=n_folds, shuffle=True, random_state=RANDOM_STATE)
    oof = np.full(len(X_fit), np.nan)

    for fold, (train_idx, valid_idx) in enumerate(folds.split(X_fit, y_fit)):
        model = TabPFNClassifier.create_default_for_version(
            "v3.5",
            n_estimators=8,
            random_state=RANDOM_STATE,
            balance_probabilities=False,
        )
        model.fit(X_fit.iloc[train_idx], y_fit.iloc[train_idx])
        oof[valid_idx] = model.predict_proba(X_fit.iloc[valid_idx])[
            :, list(model.classes_).index(1)
        ]
        print(f"TabPFN {training_size:,} rows: validation fold {fold + 1}/{n_folds}.")

    jobs_oof_predictions[training_size] = oof
```

```python
from scipy.optimize import minimize_scalar
from scipy.special import expit, logit
from sklearn.metrics import precision_score


def temperature_scale(probabilities, temperature):
    logits = logit(np.clip(probabilities, 1e-6, 1 - 1e-6))
    return expit(logits / temperature)


def fit_temperature(y_valid, probabilities):
    result = minimize_scalar(
        lambda log_temperature: log_loss(
            y_valid,
            temperature_scale(probabilities, np.exp(log_temperature)),
            labels=[0, 1],
        ),
        bounds=(np.log(0.1), np.log(10)),
        method="bounded",
    )
    return float(np.exp(result.x))


def select_f1_threshold(y_valid, probabilities):
    precision, recall, thresholds = precision_recall_curve(y_valid, probabilities)
    f1 = 2 * precision[:-1] * recall[:-1] / np.maximum(
        precision[:-1] + recall[:-1], 1e-12
    )
    best = np.flatnonzero(np.isclose(f1, f1.max()))[-1]
    return float(thresholds[best])


def score_predictions(y_true, probabilities, threshold):
    labels = probabilities >= threshold
    return {
        "Accuracy": accuracy_score(y_true, labels),
        "Precision": precision_score(y_true, labels, zero_division=0),
        "Recall": recall_score(y_true, labels, zero_division=0),
        "F1": f1_score(y_true, labels, zero_division=0),
        "AUROC": roc_auc_score(y_true, probabilities),
        "Average precision": average_precision_score(y_true, probabilities),
        "Log loss": log_loss(y_true, probabilities, labels=[0, 1]),
        "Brier": brier_score_loss(y_true, probabilities),
        "False positives": int(np.sum(labels & (np.asarray(y_true) == 0))),
    }
```
